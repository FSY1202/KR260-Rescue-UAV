cmake_minimum_required(VERSION 3.8)
project(kr260_ego_pkg)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp px4_msgs nav_msgs geometry_msgs std_msgs traj_utils kr260_ego_core kr260_ego_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(kr260_ego_node src/kr260_ego_node.cpp)
target_compile_options(kr260_ego_node PRIVATE -Wall -Wextra -O2)
ament_target_dependencies(kr260_ego_node rclcpp px4_msgs nav_msgs geometry_msgs std_msgs traj_utils kr260_ego_core kr260_ego_interfaces)
install(TARGETS kr260_ego_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/kr260_ego_runtime.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/vision_status_bridge.py scripts/servo_drop_executor.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/rescue_protocol.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
